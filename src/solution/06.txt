import { useCallback, useEffect, useMemo, useState } from "react";

const STORAGE_KEY = "about_notifications_v1";
const MAX_CLICKS = 10;

const DEFAULT_STATE = {
  notifications: true,
  clickCount: 0,
  lastState: "Be",
  lastChangedAt: null,
};

function loadInitial() {
  if (typeof window === "undefined") return DEFAULT_STATE;

  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);
    const saved = raw ? JSON.parse(raw) : null;
    if (!saved) return DEFAULT_STATE;

    const notifications = !!saved.notifications;
    const clickCount = Number(saved.clickCount || 0);
    const lastChangedAt = saved.lastChangedAt ?? null;

    // ha már elérte a limitet, legyen biztosan Ki
    const finalNotifications = clickCount >= MAX_CLICKS ? false : notifications;

    return {
      notifications: finalNotifications,
      clickCount,
      lastState: finalNotifications ? "Be" : "Ki",
      lastChangedAt,
    };
  } catch {
    return DEFAULT_STATE;
  }
}

export default function About() {
  const [notifications, setNotifications] = useState(() => loadInitial().notifications);
  const [clickCount, setClickCount] = useState(() => loadInitial().clickCount);
  const [lastState, setLastState] = useState(() => loadInitial().lastState);
  const [lastChangedAt, setLastChangedAt] = useState(() => loadInitial().lastChangedAt);

  // Mentés: csak külső rendszer frissítés
  useEffect(() => {
    if (typeof window === "undefined") return;
    try {
      window.localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify({ notifications, clickCount, lastState, lastChangedAt })
      );
    } catch {}
  }, [notifications, clickCount, lastState, lastChangedAt]);

  const lastChangedLabel = useMemo(() => {
    if (!lastChangedAt) return "—";
    const d = new Date(lastChangedAt);
    return Number.isNaN(d.getTime()) ? "—" : d.toLocaleString("hu-HU");
  }, [lastChangedAt]);

  const toggle = useCallback(() => {
    // egyetlen “tranzakcióban” frissíts mindent
    setClickCount((prevClicks) => {
      if (prevClicks >= MAX_CLICKS) return prevClicks;

      const nextClicks = prevClicks + 1;

      setNotifications((prevNotif) => {
        const nextNotif = !prevNotif;
        const finalNotif = nextClicks >= MAX_CLICKS ? false : nextNotif;

        setLastState(finalNotif ? "Be" : "Ki");
        setLastChangedAt(Date.now());

        return finalNotif;
      });

      return nextClicks;
    });
  }, []);

  const reset = useCallback(() => {
    setNotifications(true);
    setClickCount(0);
    setLastState("Be");
    setLastChangedAt(null);
  }, []);

  useEffect(() => {
    if (typeof window === "undefined") return;

    const onKeyDown = (e) => {
      const tag = window.document?.activeElement?.tagName?.toLowerCase?.();
      if (tag === "input" || tag === "textarea" || tag === "select") return;

      const key = e.key?.toLowerCase?.();
      if (key === "t") toggle();
      if (key === "r") reset();
    };

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [toggle, reset]);

  return (
    <div className="about-card">
      <p className={`status ${notifications ? "on" : "off"}`}>
        {notifications ? "Értesítések be vannak kapcsolva" : "Értesítések ki vannak kapcsolva"}
      </p>

      <div className="meta">
        <p className="meta-item">Kattintások száma: <b>{clickCount}</b></p>
        <p className="meta-item">Utolsó állapot: <b>{lastState}</b></p>
        <p className="meta-item">Utolsó váltás: <b>{lastChangedLabel}</b></p>
      </div>

      <div className="actions">
        <button className="toggle" onClick={toggle} disabled={clickCount >= MAX_CLICKS}>
          {notifications ? "Kikapcsolás" : "Bekapcsolás"} <span className="hint">(T)</span>
        </button>

        <button className="reset" onClick={reset}>
          Visszaállítás <span className="hint">(R)</span>
        </button>
      </div>

      {clickCount >= 5 && clickCount < MAX_CLICKS && (
        <p className="warning">⚠️ Túl sok kattintás!</p>
      )}

      {clickCount >= MAX_CLICKS && (
        <p className="warning strong">⛔ Elérted a limitet (10). Automatikusan kikapcsoltuk.</p>
      )}
    </div>
  );
}
